events = []
def chosen():
    events.append("chosen")
    return 7
def poison():
    events.append("poison")
    return 99
def replacement():
    events.append("replacement")
    return 8
incoming_key = "callback"
class Payload:
    def __init__(self):
        self.data = {"callback": chosen, "unused": poison}
    def keys(self):
        return [incoming_key]
    def __getitem__(self, name):
        events.append(name)
        return self.data[name]
def consume(**arguments):
    for callback in arguments.values():
        callback()
    return "handled"
payload = Payload()
result = consume(**payload)
